function renderWorkouts() { const grid = document.getElementById("workoutGrid"); grid.innerHTML = ""; workouts.forEach(workout => { const box = document.createElement("div"); box.className = "workout-box"; const name = document.createElement("input"); name.className = "workout-name-input"; name.type = "text"; name.value = workout.name; name.placeholder = "Nome do treino"; name.maxLength = 25; name.setAttribute("aria-label", "Nome da ficha"); // Impede que tocar no nome abra a ficha name.addEventListener("click", event => { event.stopPropagation(); }); name.addEventListener("pointerdown", event => { event.stopPropagation(); }); name.addEventListener("touchstart", event => { event.stopPropagation(); }); name.addEventListener("keydown", event => { event.stopPropagation(); if (event.key === "Enter") { name.blur(); } }); // Salva o nome enquanto você digita name.addEventListener("input", () => { workout.name = name.value; saveWorkouts(); }); // Organiza o nome ao terminar de editar name.addEventListener("blur", () => { workout.name = name.value.trim().toUpperCase() || "SEM NOME"; name.value = workout.name; saveWorkouts(); }); const count = document.createElement("div"); count.className = "workout-count"; count.textContent = workout.exercises.length + (workout.exercises.length === 1 ? " EXERCÍCIO" : " EXERCÍCIOS"); const arrow = document.createElement("div"); arrow.className = "workout-arrow"; arrow.textContent = "↗"; box.append(name, count, arrow); // Tocar fora do campo abre o treino box.addEventListener("click", event => { if (event.target.closest(".workout-name-input")) { return; } openWorkout(workout.id); }); grid.appendChild(box); }); }
